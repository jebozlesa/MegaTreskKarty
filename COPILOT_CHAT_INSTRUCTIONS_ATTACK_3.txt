```javascript
function executeHeal(attackerCard, defenderCard) {
  // 1. Calculate heal amount
  const randomHeal = Math.floor(Math.random() * 2) + 1; // 1-2
  const knowledgeBonus = Math.floor(attackerCard.knowledge / 4);
  const healAmount = randomHeal + knowledgeBonus;
  
  console.log(`[executeHeal] ${attackerCard.name} heals: ${healAmount} HP`);
  
  // 2. Apply heal to ATTACKER (self-heal)
  attackerCard.health += healAmount;
  if (attackerCard.health > attackerCard.maxHealth) {
    attackerCard.health = attackerCard.maxHealth;
  }
  
  // 3. Remove negative effects (IDs: 1, 4, 13, 24)
  const effectsToRemove = [1, 4, 13, 24];
  const effectsRemoved = [];
  
  if (attackerCard.effects && attackerCard.effects.length > 0) {
    const originalEffects = [...attackerCard.effects];
    attackerCard.effects = attackerCard.effects.filter(effect => {
      // Assuming effect has { type: number } or { id: number }
      const effectId = effect.type || effect.id;
      if (effectsToRemove.includes(effectId)) {
        effectsRemoved.push(effectId);
        return false; // Remove this effect
      }
      return true; // Keep this effect
    });
    
    if (effectsRemoved.length > 0) {
      console.log(`[executeHeal] Removed effects: ${effectsRemoved.join(', ')}`);
    }
  }
  
  // 4. Return result
  return {
    healAmount: healAmount,
    effectsRemoved: effectsRemoved,
    damage: 0  // No damage dealt
  };
}
```

NOTE: Effect structure depends on your current implementation.
Adjust "effect.type" vs "effect.id" based on existing code.
